cmake_minimum_required(VERSION 3.24)
project(xgcrc_consumer LANGUAGES C)

set(XGCRC_CONSUMER_MODE "both" CACHE STRING "crc8, crc16, both, optional, unknown or wrong_version")
set(XGCRC_CONSUMER_SOURCE_DIR "" CACHE PATH "Explicit source checkout; empty uses an installed package")
option(XGCRC_CONSUMER_CHECK_CXX "Build a separate C++ header/link consumer" OFF)

if(XGCRC_CONSUMER_MODE STREQUAL "crc8" OR XGCRC_CONSUMER_MODE STREQUAL "optional")
    set(components crc8)
elseif(XGCRC_CONSUMER_MODE STREQUAL "crc16")
    set(components crc16)
elseif(XGCRC_CONSUMER_MODE STREQUAL "unknown")
    set(components crc32)
else()
    set(components crc8 crc16)
endif()

if(XGCRC_CONSUMER_SOURCE_DIR)
    set(XGCRC_BUILD_TESTS OFF CACHE BOOL "Consumer disables upstream tests" FORCE)
    set(XGCRC_ENABLE_COVERAGE OFF CACHE BOOL "Consumer uses production libraries" FORCE)
    foreach(component crc8 crc16)
        string(TOUPPER "${component}" upper_component)
        if(component IN_LIST components)
            set(XGCRC_BUILD_${upper_component} ON CACHE BOOL "Selected algorithm" FORCE)
        else()
            set(XGCRC_BUILD_${upper_component} OFF CACHE BOOL "Unselected algorithm" FORCE)
        endif()
    endforeach()
    add_subdirectory("${XGCRC_CONSUMER_SOURCE_DIR}" dependency)
elseif(XGCRC_CONSUMER_MODE STREQUAL "optional")
    find_package(xgen_crc 0.1.0 EXACT CONFIG REQUIRED COMPONENTS crc8 OPTIONAL_COMPONENTS crc32)
    if(xgen_crc_crc32_FOUND)
        message(FATAL_ERROR "Unknown optional component must not be reported as available")
    endif()
elseif(XGCRC_CONSUMER_MODE STREQUAL "wrong_version")
    find_package(xgen_crc 0.2.0 EXACT CONFIG REQUIRED COMPONENTS crc8)
else()
    find_package(xgen_crc 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
endif()

set(consume_crc8 0)
set(consume_crc16 0)
set(consumer_targets "")
foreach(component crc8 crc16)
    if(component IN_LIST components)
        if(NOT TARGET xgcrc::${component})
            message(FATAL_ERROR "Requested target xgcrc::${component} is missing")
        endif()
        get_target_property(version xgcrc::${component} XGCRC_VERSION)
        get_target_property(abi xgcrc::${component} XGCRC_ABI_VERSION)
        if(NOT version VERSION_EQUAL "0.1.0" OR NOT abi STREQUAL "1")
            message(FATAL_ERROR "Component version/ABI properties were not preserved")
        endif()
        get_target_property(dependencies xgcrc::${component} INTERFACE_LINK_LIBRARIES)
        if(dependencies)
            message(FATAL_ERROR "CRC algorithms must not have library dependencies: ${dependencies}")
        endif()
        list(APPEND consumer_targets xgcrc::${component})
        set(consume_${component} 1)
    elseif(TARGET xgcrc::${component})
        message(FATAL_ERROR "An unselected CRC algorithm was imported or built")
    endif()
endforeach()
if(TARGET GTest::gtest OR TARGET GTest::gtest_main OR TARGET xgc::base)
    message(FATAL_ERROR "Production consumers must not import GoogleTest or core")
endif()

add_executable(consumer_c consumer.c)
set_target_properties(consumer_c PROPERTIES
    C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer_c PRIVATE ${consumer_targets})
target_compile_definitions(consumer_c PRIVATE
    XGCRC_CONSUME_CRC8=${consume_crc8} XGCRC_CONSUME_CRC16=${consume_crc16})
enable_testing()
add_test(NAME consumer_c_runs COMMAND consumer_c)

if(XGCRC_CONSUMER_CHECK_CXX)
    enable_language(CXX)
    add_executable(consumer_cpp consumer.cpp)
    set_target_properties(consumer_cpp PROPERTIES
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_link_libraries(consumer_cpp PRIVATE ${consumer_targets})
    target_compile_definitions(consumer_cpp PRIVATE
        XGCRC_CONSUME_CRC8=${consume_crc8} XGCRC_CONSUME_CRC16=${consume_crc16})
    add_test(NAME consumer_cpp_runs COMMAND consumer_cpp)
endif()
